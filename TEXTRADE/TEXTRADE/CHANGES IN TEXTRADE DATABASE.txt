--CHANGE THE YEARID MANUALLY AND EXECUTE
begin
declare @OLDGDNNO VARCHAR(50), @INVINITIALS VARCHAR(50), @SPOLDGDNNO INT, @STROLDGDNNO INT, @YEARID INT
SET @YEARID = 1
declare CUR_MAIN cursor for select INVOICE_GDNNO ,INVOICE_INITIALS from INVOICEMASTER where INVOICE_YEARID = @YEARID ORDER BY INVOICE_REGISTERID, INVOICE_NO
open CUR_MAIN
	fetch next from CUR_MAIN into @OLDGDNNO,@INVINITIALS
	while @@Fetch_STATUS = 0
		begin
			
			while @OLDGDNNO	<> '' 
			begin					
				
				SET @SPOLDGDNNO = CHARINDEX(',', @OLDGDNNO) 
				IF @spOLDGDNNO>0 
					BEGIN 
						SET @strOLDGDNNO = CAST(LEFT(@OLDGDNNO, @spOLDGDNNO-1) AS INT) 
						SET @OLDGDNNO = RIGHT(@OLDGDNNO, LEN(@OLDGDNNO)-@spOLDGDNNO) 
					END 
				ELSE 
					BEGIN 
						SET @strOLDGDNNO = CAST(@OLDGDNNO AS INT) 
						SET @OLDGDNNO = '' 
					END 
				UPDATE GDN SET GDN_INVOICEINITIALS = @INVINITIALS  WHERE GDN_YEARID = @YEARID AND GDN_NO = @STROLDGDNNO
			END

		fetch next from CUR_MAIN into @OLDGDNNO,@INVINITIALS
		end
close CUR_MAIN
deallocate CUR_MAIN
end
